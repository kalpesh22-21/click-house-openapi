+class QueryRowsStream:
+    """Context-managed row stream over a ClickHouse result, with column metadata.
+
+    Wraps the streaming ``QueryResult`` so the caller gets
+    ``column_names`` / ``column_type_names`` up front AND iterates rows lazily
+    WITHOUT materializing the result set (clickhouse-connect's public
+    ``query_rows_stream`` discards the QueryResult that carries the columns, so
+    this wrapper keeps it).  MUST be used in a ``with`` block so the underlying
+    HTTP byte source is closed even on abort — leaking it wedges the pooled
+    connection.
+
+    Iteration raises the SAME HTTPException(400)/HTTPException(502) shapes
+    execute_query raises: a ClickHouse failure embedded in the Native stream
+    (e.g. a ``result_overflow_mode='throw'`` overflow) surfaces mid-iteration as
+    a DatabaseError and is translated identically.
+    """
+
+    def __init__(self, query_result: Any, settings: Settings) -> None:
+        self._query_result = query_result
+        self._settings = settings
+        self.column_names: list[str] = list(query_result.column_names)
+        self.column_type_names: list[str] = [
+            t.name for t in query_result.column_types
+        ]
+        self._stream_cm: Any = None
+
+    def __enter__(self) -> "QueryRowsStream":
+        self._stream_cm = self._query_result.rows_stream
+        self._stream_cm.__enter__()
+        return self
+
+    def __exit__(self, *exc_info: Any) -> Any:
+        stream_cm, self._stream_cm = self._stream_cm, None
+        return stream_cm.__exit__(*exc_info)
+
+    def __iter__(self):
+        if self._stream_cm is None:
+            raise RuntimeError(
+                "QueryRowsStream must be entered (with-block) before iteration."
+            )
+        try:
+            for row in self._stream_cm:
+                yield list(row)
+        except DatabaseError as exc:
+            raise _query_error_http(exc, self._settings) from exc
+        except ClickHouseError as exc:
+            raise _unavailable_http(exc, self._settings) from exc

+class SnapshotObjectMissingError(Exception):
+    """The requested object (or prefix) does not exist in the bucket.
+
+    Raised by read methods on a genuine miss.  The service layer maps this to
+    SNAPSHOT_NOT_FOUND — it never distinguishes a foreign-tenant id from an
+    unknown id, and this single exception type keeps that boundary clean.
+    """

+class S3SnapshotStore:
+    """boto3-backed store for snapshot objects under a configured bucket/prefix.
+
+    All methods that talk to S3 may raise ``botocore.exceptions.ClientError``
+    (translated to :class:`SnapshotObjectMissingError` on a missed read) or a
+    network-level ``BotoCoreError``; the service layer treats any such failure
+    as an upload/read failure and cleans up/raises accordingly.
+    """
+
+    def __init__(self, settings: Settings) -> None:
+        # Lazy SDK imports (see module docstring): an EXPORT_STORE=couchbase
+        # deployment imports this module only for SnapshotObjectMissingError.
+        import boto3
+        from botocore.client import Config
+
+        self._bucket = settings.export_s3_bucket
+        self._client = boto3.client(
+            "s3",
+            endpoint_url=settings.export_s3_endpoint or None,
+            region_name=settings.export_s3_region,
+            aws_access_key_id=settings.export_s3_access_key or None,
+            aws_secret_access_key=settings.export_s3_secret_key or None,
+            # Path-style addressing is REQUIRED by the on-prem targets (Dell ECS
+            # / NetApp StorageGRID); virtual-hosted style would build hosts like
+            # <bucket>.<endpoint> that their TLS/DNS does not serve.  SigV4 is
+            # pinned explicitly: botocore presigns S3 URLs with legacy SigV2
+            # unless told otherwise, and the modern on-prem appliances expect V4.
+            config=Config(s3={"addressing_style": "path"}, signature_version="s3v4"),
+        )
+
+    @property
+    def bucket(self) -> str:
+        return self._bucket
+
+    def put_file(self, local_path: str, key: str) -> None:
+        """Upload the staged file at *local_path* to ``s3://bucket/key``."""
+        self._client.upload_file(local_path, self._bucket, key)

+    def read_bytes(self, key: str) -> bytes:
+        """Return the full object body (small objects only — meta.json)."""
+        from botocore.exceptions import ClientError

+            return self._client.get_object(Bucket=self._bucket, Key=key)["Body"]
+        except ClientError as exc:

+    """Encode one streamed row as its REST-wire JSON values.
+
+    NUL-clean first, then jsonable_encoder — this matches what FastAPI/pydantic
+    emits for /query rows today (datetime/date → ISO strings, Decimal → number,
+    tuples → arrays), so a snapshot's rows are byte-identical in value to the
+    rows the interactive path would have returned.
+    """
+    cleaned = _clean_string_padding(list(row))
+    return jsonable_encoder(cleaned)
+
+
+def _csv_cell(value: Any) -> str:
+    """Render one wire value as a canonical CSV cell (display formatting stays UI-side)."""
+    if value is None:
+        return ""
+    if isinstance(value, bool):
+        return "true" if value else "false"
+    if isinstance(value, (int, float)):
+        return str(value)
+    if isinstance(value, str):
+        return value
+    # Arrays / maps / other composites: canonical compact JSON.
+    return json.dumps(value, ensure_ascii=False, separators=(",", ":"))
+
+
+def _csv_line(values: list[str]) -> str:
+    """Serialize one CSV record exactly as written to the part file."""
+    buf = io.StringIO()
+    csv.writer(buf, lineterminator="\r\n").writerow(values)
+    return buf.getvalue()
+
+
+class _JsonlGzRowsWriter:
+    """S3-mode rows sink: stage rows as gzip JSON Lines on local disk.
+
+    One line per row, a JSON array of wire values; the staged file is uploaded
+    verbatim to ``rows.jsonl.gz`` after the stream completes (the paging source
+    of truth).  Text-mode gzip, ``newline='\\n'`` — no platform translation.
+    """
+
+    def __init__(self, tmpdir: str) -> None:
+        self.rows_path = f"{tmpdir}/rows.jsonl.gz"
+        self._jf = None
+
+    def write(self, wire_row: list[Any], line: str) -> None:
+        if self._jf is None:
+            self._jf = gzip.open(self.rows_path, "wt", encoding="utf-8", newline="\n")
+        self._jf.write(line)
+
+    def finish(self) -> None:
+        if self._jf is None:
+            # A zero-row snapshot never opened the JSONL stream lazily — still
+            # stage an (empty) stream so paging reads always find the object.
+            self._jf = gzip.open(self.rows_path, "wt", encoding="utf-8", newline="\n")
+        self._jf.close()
+        self._jf = None
+
+
+class _ChunkedRowsDocsWriter:
+    """Couchbase-mode rows sink: persist row-chunk JSON docs DURING streaming.
+
+    Chunks are 0-indexed (``rows.0000``, ``rows.0001``, …) and hold up to
+    EXPORT_CB_CHUNK_ROWS wire rows apiece, so no monolithic rows object is ever
+    staged and paging fetches O(1) docs by ceil-math.  The chunk docs inherit
+    the snapshot's per-doc expiry from the store.
+    """
+
+    def __init__(self, store: Any, prefix: str, chunk_rows: int) -> None:
+        self._store = store
+        self._prefix = prefix
+        self._chunk_rows = chunk_rows
+        self._buf: list[list[Any]] = []
+        self._next_chunk = 0
+
+    def write(self, wire_row: list[Any], line: str) -> None:
+        # ``line`` is the S3 sink's pre-encoded JSONL text; the docs sink stores
+        # the row itself and ignores line entirely.
+        self._buf.append(wire_row)
+        if len(self._buf) >= self._chunk_rows:
+            self._flush()
+
+    def _flush(self) -> None:
+        key = CouchbaseSnapshotStore.rows_chunk_key(self._prefix, self._next_chunk)
+        self._store.put_json(key, self._buf)
+        self._next_chunk += 1
+        self._buf = []
+
+    def finish(self) -> None:
+        if self._buf:
+            self._flush()
+
+
+class _Staging:
+    """Local temp-file staging for one snapshot while rows stream in.
+
+    Everything stays on local disk under a TemporaryDirectory until the stream
+    COMPLETES; uploads happen only after that, so a budget abort or a mid-stream
+    ClickHouse failure leaves zero orphan artifacts in the store (Couchbase-mode
+    row chunks are the deliberate exception: they are streamed docs and are
+    reaped by delete_prefix on abort / per-doc expiry otherwise).  Byte counters
+    here are the authoritative budget metric: JSONL bytes are counted as
+    written PRE-compression (a stable upper proxy for the stored size) and CSV
+    bytes exactly.
+    """
+
+    def __init__(self, settings: Settings, columns: list[str], rows_writer_factory: Any) -> None:
+        self._settings = settings
+        self._columns = columns
+        self._tmpdir = tempfile.TemporaryDirectory(prefix="chapi-export-")
+        self.rows = rows_writer_factory(self._tmpdir.name)
+        self.parts: list[dict[str, Any]] = []  # {path, name, rows, bytes} staged CSV parts
+        self.total_rows = 0
+        self.total_bytes = 0
+        self._part_fh = None
+        self._part_rows = 0
+        self._part_bytes = 0
+        self._finished = False
+
+    def _start_part(self, part_no: int) -> None:
+        name = f"data.part-{part_no:04d}.csv"
+        path = f"{self._tmpdir.name}/{name}"
+        # newline='' keeps the csv writer's own \r\n terminators untranslated.
+        self._part_fh = open(path, "w", encoding="utf-8", newline="")
+        header = _csv_line(self._columns)
+        self._part_fh.write(header)
+        self._part_bytes = len(header.encode("utf-8"))
+        self._part_rows = 0
+        self.parts.append({"path": path, "name": name, "rows": 0, "bytes": self._part_bytes})
+
+    def _close_part(self) -> None:
+        if self._part_fh is None:
+            return
+        self.parts[-1]["rows"] = self._part_rows
+        self.parts[-1]["bytes"] = self._part_bytes
+        self._part_fh.close()
+        self._part_fh = None
+
+    def write_row(self, wire_row: list[Any]) -> None:
+        """Write one wire row to the rows sink and the current CSV part.
+
+        Raises ExportTooLargeError BEFORE writing when the row would exceed a
+        total budget — the snapshot is then aborted.
+        """
+        jsonl_line = json.dumps(wire_row, ensure_ascii=False, separators=(",", ":")) + "\n"
+        jsonl_bytes = len(jsonl_line.encode("utf-8"))
+        csv_line = _csv_line([_csv_cell(v) for v in wire_row])
+        csv_bytes = len(csv_line.encode("utf-8"))
+
+        if (
+            self.total_rows + 1 > self._settings.export_total_max_rows
+            or self.total_bytes + jsonl_bytes + csv_bytes > self._settings.export_total_max_bytes
+        ):
+            raise ExportTooLargeError(
+                message=(
+                    "This export exceeds the maximum snapshot size "
+                    f"({self._settings.export_total_max_rows} rows / "
+                    f"{self._settings.export_total_max_bytes // (1024 * 1024)} MB). "
+                    "Narrow the result with a more selective WHERE clause or a "
+                    "smaller LIMIT and export again."
+                ),
+                code="EXPORT_TOO_LARGE",
+            )
+
+        if self._part_fh is None or (
+            self._part_rows >= self._settings.export_max_rows
+            or self._part_bytes + csv_bytes > self._settings.export_part_max_bytes
+        ):
+            self._close_part()
+            self._start_part(len(self.parts) + 1)
+
+        self.rows.write(wire_row, jsonl_line)
+        self._part_fh.write(csv_line)
+        self._part_rows += 1
+        self._part_bytes += csv_bytes
+        self.total_rows += 1
+        self.total_bytes += jsonl_bytes + csv_bytes
+
+    def write_meta(self, meta: dict[str, Any]) -> str:
+        """Stage meta.json locally and return its path (uploaded LAST by the caller)."""
+        meta_path = f"{self._tmpdir.name}/meta.json"
+        with open(meta_path, "w", encoding="utf-8") as fh:
+            json.dump(meta, fh, separators=(",", ":"))
+        return meta_path
+
+    def finish(self) -> None:
+        """Close open staged artifacts (part list is left truthful for meta)."""
+        if self._finished:
+            return
+        self._close_part()
+        self.rows.finish()
+        self._finished = True
+
+    def cleanup(self) -> None:
+        try:
+            self.finish()
+        finally:
+            self._tmpdir.cleanup()
+
+    def __enter__(self) -> _Staging:
+        return self
+
+    def __exit__(self, *exc_info: object) -> None:
+        self.cleanup()
+

+def create_snapshot(
+    sql: str,
+    settings: Settings | None = None,
+    *,
+    store: Any = None,
+) -> dict[str, Any]:
+    """Mint an answer snapshot: run *sql* to completion and persist it.
+
+    Guardrail chain is IDENTICAL to service.run_query (validation → shared
+    guardrails → employee-access freshness), with inject_limit=False as the only
+    deliberate difference: snapshots hold the FULL result set.
+
+    Returns the create-response shape: {snapshot_id, columns, row_count,
+    created_at, expires_at} with column NAMES only (matches runQuery's compact
+    shape).
+    """
+    if settings is None:
+        settings = get_settings()
+
+    try:
+        clean_sql = validate_and_sanitize(sql, settings.default_limit, inject_limit=False)
+    except HTTPException as exc:
+        raise QueryValidationError(
+            message=exc.detail.get("error", "Query validation failed"),
+            code=exc.detail.get("code", "QUERY_VALIDATION_ERROR"),
+        ) from exc
+
+    # Same shared guardrails run_query applies, same flags (REST callers are
+    # unscoped — provenance triggers only for scratch references, exactly like
+    # /query). Nothing about the export path weakens them.
+    _enforce_query_guardrails(
+        clean_sql, settings, caller="export_snapshot", include_provenance=False
+    )
+
+    # RLS freshness is frozen at MINT time: the row-policy set is refreshed here
+    # (once) and never re-checked on page/download (those verify only the tenant
+    # hash and the expiry clock).
+    ensure_employee_access_fresh(settings)
+
+    tenant = tenant_settings(settings)
+    thash = _tenant_hash(tenant)
+    store = _resolve_store(settings, store)
+    store_mode = settings.export_store
+
+    snapshot_id = str(uuid.uuid4())
+    created_at = datetime.now(timezone.utc)
+    expires_at = created_at + timedelta(seconds=settings.export_ttl_seconds)
+    prefix = _artifact_prefix(settings, thash, snapshot_id)
+
+    caps = export_settings(settings)
+    staged_keys: list[str] = []
+    try:
+        stream = execute_query_rows_stream(clean_sql, settings, caps=caps)
+        with stream as rows:
+            columns = stream.column_names
+            column_types = stream.column_type_names
+            if store_mode == "couchbase":
+                def rows_writer_factory(tmpdir: str) -> Any:
+                    return _ChunkedRowsDocsWriter(
+                        store, prefix, settings.export_cb_chunk_rows
+                    )
+            else:
+                rows_writer_factory = _JsonlGzRowsWriter
+            with _Staging(settings, columns, rows_writer_factory) as staging:
+                for row in rows:
+                    staging.write_row(_wire_row(row))
+                staging.finish()
+
+                # Stream complete — now upload. meta goes LAST (commit point).
+                parts_meta = []
+                if store_mode == "couchbase":
+                    for index, part in enumerate(staging.parts, start=1):
+                        key = CouchbaseSnapshotStore.csv_part_key(prefix, index)
+                        store.put_file(part["path"], key)
+                        staged_keys.append(key)
+                        parts_meta.append(
+                            {"key": key, "rows": part["rows"], "bytes": part["bytes"]}
+                        )
+                else:
+                    jsonl_key = f"{prefix}/rows.jsonl.gz"
+                    store.put_file(staging.rows.rows_path, jsonl_key)
+                    staged_keys.append(jsonl_key)
+
+                    for part in staging.parts:
+                        key = f"{prefix}/{part['name']}"
+                        store.put_file(part["path"], key)
+                        staged_keys.append(key)
+                        parts_meta.append(
+                            {"key": key, "rows": part["rows"], "bytes": part["bytes"]}
+                        )
+
+                meta = {
+                    "snapshot_id": snapshot_id,
+                    "tenant_hash": thash,
+                    "columns": [
+                        {"name": name, "type": ctype}
+                        for name, ctype in zip(columns, column_types)
+                    ],
+                    "row_count": staging.total_rows,
+                    "created_at": _iso_z(created_at),
+                    "expires_at": _iso_z(expires_at),
+                    "parts": parts_meta,
+                    "sql_sha256": hashlib.sha256(clean_sql.encode("utf-8")).hexdigest(),
+                }
+                if store_mode == "couchbase":
+                    meta["chunk_rows"] = settings.export_cb_chunk_rows
+                    meta_key = CouchbaseSnapshotStore.meta_key(prefix)
+                    store.put_json(meta_key, meta)
+                else:
+                    meta_key = f"{prefix}/meta.json"
+                    store.put_file(staging.write_meta(meta), meta_key)
+                staged_keys.append(meta_key)
+    except HTTPException as exc:
+        # ClickHouse query/connectivity failure from the stream boundary —
+        # delete anything already uploaded, then translate to the service-layer
+        # domain errors (same shapes run_query raises) for the router to map.
+        _best_effort_delete(store, prefix, snapshot_id)
+        if exc.status_code == 502:
+            raise ClickHouseUnavailableError(
+                message=exc.detail.get("error", "ClickHouse unavailable"),
+                code=exc.detail.get("code", "CLICKHOUSE_UNAVAILABLE"),
+            ) from exc
+        raise ClickHouseQueryError(
+            message=exc.detail.get("error", "ClickHouse query error"),
+            code=exc.detail.get("code", "CLICKHOUSE_QUERY_ERROR"),
+        ) from exc
+    except Exception:
+        # Budget abort, storage failure, any mid-upload breakage: delete whatever
+        # managed to upload so a half-written snapshot never becomes readable.
+        _best_effort_delete(store, prefix, snapshot_id)
+        raise
+
+    logger.info(
+        "snapshot created id=%s rows=%d parts=%d",
+        snapshot_id,
+        staging.total_rows,
+        len(parts_meta),
+    )
+    return {
+        "snapshot_id": snapshot_id,
+        "columns": columns,
+        "row_count": staging.total_rows,
+        "created_at": _iso_z(created_at),
+        "expires_at": _iso_z(expires_at),
+    }
+
+
+def _best_effort_delete(store: Any, prefix: str, snapshot_id: str) -> None:
+    try:
+        store.delete_prefix(prefix)
+    except Exception:  # noqa: BLE001 — cleanup must never mask the real error
+        logger.warning(
+            "snapshot cleanup failed id=%s (orphan objects expire via bucket lifecycle)",
+            snapshot_id,
+        )
+
+
+def _load_meta(
+    store: Any, settings: Settings, snapshot_id: str,
+) -> dict[str, Any]:
+    """Read and authorize the meta artifact for *snapshot_id* from the CURRENT tenant.
+
+    Raises SnapshotNotFoundError for: malformed id, missing doc/object, corrupt
+    metadata, OR a tenant-hash mismatch — all four are deliberately the same wire
+    error so existence is never probeable across tenants.
+    """
+    if not _SNAPSHOT_ID_RE.match(snapshot_id):
+        raise SnapshotNotFoundError(
+            message="Snapshot not found.", code="SNAPSHOT_NOT_FOUND"
+        )
+    thash = _tenant_hash(tenant_settings(settings))
+    meta_key = _meta_key(settings, thash, snapshot_id)
+    try:
+        raw = store.read_bytes(meta_key)
+    except SnapshotObjectMissingError as exc:
+        raise SnapshotNotFoundError(
+            message="Snapshot not found.", code="SNAPSHOT_NOT_FOUND"
+        ) from exc
+    try:
+        meta = json.loads(raw.decode("utf-8"))
+    except (ValueError, UnicodeDecodeError) as exc:
+        logger.error("snapshot meta unreadable id=%s", snapshot_id)
+        raise SnapshotNotFoundError(
+            message="Snapshot not found.", code="SNAPSHOT_NOT_FOUND"
+        ) from exc
+    if not hmac.compare_digest(str(meta.get("tenant_hash", "")), thash):
+        # Foreign-tenant id: identical wire error to a missing snapshot.
+        raise SnapshotNotFoundError(
+            message="Snapshot not found.", code="SNAPSHOT_NOT_FOUND"
+        )
+    return meta
+
+
+def _check_expiry(meta: dict[str, Any]) -> None:
+    """410 SNAPSHOT_EXPIRED once the TTL clock runs out (app-side enforcement)."""
+    try:
+        expires_at = _parse_iso_z(str(meta["expires_at"]))
+    except (KeyError, ValueError) as exc:
+        logger.error("snapshot meta missing/invalid expires_at id=%s", meta.get("snapshot_id"))
+        raise SnapshotNotFoundError(
+            message="Snapshot not found.", code="SNAPSHOT_NOT_FOUND"
+        ) from exc
+    if datetime.now(timezone.utc) >= expires_at:
+        raise SnapshotExpiredError(
+            message="This snapshot has expired. Refresh the data to mint a new one.",
+            code="SNAPSHOT_EXPIRED",
+        )
+
+
+def _iter_jsonl_rows(reader: Any) -> Iterator[list[Any]]:
+    """Yield stored rows lazily from a rows.jsonl.gz object stream."""
+    with gzip.GzipFile(fileobj=reader, mode="rb") as gz:
+        wrapper = io.TextIOWrapper(gz, encoding="utf-8")
+        for line in wrapper:
+            line = line.rstrip("\n")
+            if line:
+                yield json.loads(line)
+
+
+def _read_rows_window_cb(
+    store: Any,
+    settings: Settings,
+    meta: dict[str, Any],
+    snapshot_id: str,
+    offset: int,
+    limit: int,
+) -> list[list[Any]]:
+    """Couchbase-mode page read: fetch O(1) row-chunk docs and slice in memory.
+
+    Chunk doc ids derive from row offsets (chunk = offset // chunk_rows), so a
+    page never scans.  A missing chunk doc mirrors the S3 missing-payload rule:
+    the meta outlived its rows (per-doc expiry raced the clock), so the snapshot
+    is treated as EXPIRED — the UI offers Refresh, not a dead end.
+    """
+    thash = str(meta["tenant_hash"])
+    row_count = int(meta.get("row_count", 0))
+    chunk_rows = int(meta.get("chunk_rows") or settings.export_cb_chunk_rows)
+    prefix = CouchbaseSnapshotStore.prefix(thash, snapshot_id)
+
+    rows: list[list[Any]] = []
+    idx = offset
+    while len(rows) < limit and idx < row_count:
+        chunk_no, within = divmod(idx, chunk_rows)
+        key = CouchbaseSnapshotStore.rows_chunk_key(prefix, chunk_no)
+        try:
+            chunk = json.loads(store.read_bytes(key).decode("utf-8"))
+        except SnapshotObjectMissingError as exc:
+            raise SnapshotExpiredError(
+                message="This snapshot has expired. Refresh the data to mint a new one.",
+                code="SNAPSHOT_EXPIRED",
+            ) from exc
+        take = chunk[within : within + (limit - len(rows))]
+        if not take:  # defensive: a short/empty trailing chunk must not loop
+            break
+        rows.extend(take)
+        idx += len(take)
+    return rows
+
+
+def page_snapshot(
+    snapshot_id: str,
+    offset: int,
+    limit: int,
+    settings: Settings | None = None,
+    *,
+    store: Any = None,
+) -> dict[str, Any]:
+    """Return rows [offset, offset+limit) of a frozen snapshot.
+
+    ``has_more`` is EXACT: ``offset + len(rows) < row_count``.  An offset at or
+    past the end is a 200 with ``rows: []`` and ``has_more: false`` — never an
+    error.  Rows are served exactly as stored at mint time, so a page never
+    reflects data changes after mint.
+    """
+    if settings is None:
+        settings = get_settings()
+    store = _resolve_store(settings, store)
+
+    meta = _load_meta(store, settings, snapshot_id)
+    _check_expiry(meta)
+
+    row_count = int(meta.get("row_count", 0))
+    columns = [c.get("name", "") for c in meta.get("columns", [])]
+
+    rows: list[list[Any]] = []
+    if 0 <= offset < row_count and limit > 0:
+        if settings.export_store == "couchbase":
+            rows = _read_rows_window_cb(store, settings, meta, snapshot_id, offset, limit)
+        else:
+            rows_key = f"{_snapshot_prefix(settings, meta['tenant_hash'], snapshot_id)}/rows.jsonl.gz"
+            try:
+                reader = store.open_reader(rows_key)
+            except SnapshotObjectMissingError as exc:
+                # meta.json exists but the payload is gone (e.g. lifecycle reaped
+                # it ahead of the clock) — from the caller's side the snapshot
+                # cannot be served; treat as expired rather than not-found so the
+                # UI offers the Refresh path instead of a dead end.
+                raise SnapshotExpiredError(
+                    message="This snapshot has expired. Refresh the data to mint a new one.",
+                    code="SNAPSHOT_EXPIRED",
+                ) from exc
+            with reader:
+                for index, stored_row in enumerate(_iter_jsonl_rows(reader)):
+                    if index < offset:
+                        continue
+                    if len(rows) >= limit:
+                        break
+                    rows.append(stored_row)
+
+    return {
+        "columns": columns,
+        "rows": rows,
+        "limit": limit,
+        "offset": offset,
+        "has_more": offset + len(rows) < row_count,
+        "row_count": row_count,
+        "snapshot_id": snapshot_id,
+        "expires_at": meta["expires_at"],
+    }
+

+def download_urls(
+    snapshot_id: str,
+    settings: Settings | None = None,
+    *,
+    store: Any = None,
+) -> dict[str, Any]:
+    """Return the download location for a snapshot's CSV parts, per store mode.
+
+    S3 mode (byte-identical to CONTRACTS §1): ordered short-lived presigned
+    URLs, each signed with ``ResponseContentDisposition``, ``expires_in``
+    = EXPORT_PRESIGN_TTL_SECONDS.
+
+    Couchbase mode (CONTRACTS §6.2): each file's ``url`` is a RELATIVE app path
+    ``query/snapshots/{id}/download?part=N`` carrying per-request authz in place
+    of the presign window, so ``expires_in`` is ``null``.
+
+    Parts come back in order (part 1 first).  A zero-row snapshot has no parts
+    and returns ``files: []``.
+    """
+    if settings is None:
+        settings = get_settings()
+    store = _resolve_store(settings, store)
+
+    meta = _load_meta(store, settings, snapshot_id)
+    _check_expiry(meta)
+
+    if settings.export_store == "couchbase":
+        files = [
+            {
+                "part": index,
+                "url": f"query/snapshots/{snapshot_id}/download?part={index}",
+                "rows": part["rows"],
+                "filename": f"query-export-part-{index:04d}.csv",
+            }
+            for index, part in enumerate(meta.get("parts", []), start=1)
+        ]
+        return {"files": files, "expires_in": None}
+
+    expires_in = settings.export_presign_ttl_seconds
+    files = []
+    for index, part in enumerate(meta.get("parts", []), start=1):
+        filename = f"query-export-part-{index:04d}.csv"
+        url = store.presign_get(part["key"], filename=filename, expires_in=expires_in)
+        files.append(
+            {"part": index, "url": url, "rows": part["rows"], "filename": filename}
+        )
+
+    return {"files": files, "expires_in": expires_in}
+

+def download_part(
+    snapshot_id: str,
+    part: int,
+    settings: Settings | None = None,
+    *,
+    store: Any = None,
+) -> tuple[Iterator[bytes], str]:
+    """Return a (byte iterator, filename) streaming one stored CSV part.
+
+    Backs GET /query/snapshots/{id}/download?part=N — the only download
+    transport in Couchbase mode (CONTRACTS §6.2); also available in S3 mode.

+# ADR-0003: Answer snapshots — frozen full-result exports on S3-compatible storage
+
+**Status:** Accepted
+**Date:** 2026-10-22
+**Supersedes/relates:** extends ADR-0001 (per-tenant JWT row isolation) unchanged;
+the snapshot guardrail chain reuses ADR-0002's shared `_enforce_query_guardrails`.
+Ticket: V4-0101010. Full styled plan: `docs/plans/answer-snapshot-export.html`.
+
+---
+
+## Context
+
+The data-agent UI paged query results by RE-EXECUTING the SQL for every page
+(`SELECT * FROM (<sql>) LIMIT n OFFSET m`), and exported by looping the client
+over those pages with a hard 5000-row cap.  Consequences: every page is a fresh
+query (cost, and no consistency between pages or between screen and file), the
+UI's "N results" byline was a growing estimate with no true total, and exports
+were silently truncated.
+
+Locked product requirements:
+
+1. The export file always contains ALL the data — or is delivered as ordered
+   downloadable parts.
+2. The frontend pages from a server-side snapshot, never by re-running the query.
+3. Data on the UI must EXACTLY match the export file (same snapshot).
+4. "Refresh data" is the ONLY way data changes.
+5. A new answer always pulls fresh data — byte-identical SQL mints its own
+   snapshot.
+
+---
+
+## Decision
+
+1. **Separate export path; interactive `/query` untouched.**  New REST routes
+   `POST /query/snapshots`, `GET /query/snapshots/{id}/page`,
+   `GET /query/snapshots/{id}/download-urls`, registered only when
+   `EXPORT_ENABLED=true`.  No new MCP tool — REST-only surface.
+
+2. **Same guardrails, one deliberate exception.**  `create_snapshot` runs the
+   identical chain as `run_query` (`validate_and_sanitize` →
+   `_enforce_query_guardrails` → `ensure_employee_access_fresh` → execute),
+   except validation is called with the new keyword-only `inject_limit=False`:
+   the snapshot materializes the FULL result set, so the injected
+   `DEFAULT_LIMIT` is never appended.  Every other validation transform is
+   byte-identical.
+
+3. **Fail loud, never silently truncate.**  The export streams with its own
+   caps bundle (`app/clickhouse_client.export_settings`) whose values derive
+   EXCLUSIVELY from `EXPORT_*` config — never merged with the operator's
+   interactive `MAX_*` overrides — with `result_overflow_mode='throw'`
+   (interactive uses 'break').  Budgets (`EXPORT_TOTAL_MAX_ROWS` /
+   `EXPORT_TOTAL_MAX_BYTES`, defaults 1M rows / 100 MB) are enforced LOCALLY
+   per streamed row by `app/export_service.py`; an over-budget mint aborts,
+   deletes any partial objects, and raises 400 `EXPORT_TOO_LARGE` with a
+   message advising a narrower WHERE/LIMIT.  ClickHouse error text is never
+   parsed to detect overflow — the CH-side `max_result_rows` cap deliberately
+   sits ONE row above the row budget so the local counter always trips first
+   (both sides count the same streamed rows), making the domain error
+   authoritative.  CH-side byte cap is a coarse backstop only (serialized
+   result bytes ≠ staged file bytes).
+
+4. **Auto-split CSV parts; numbered, standalone, header-bearing.**  Parts split
+   at `EXPORT_MAX_ROWS` rows OR `EXPORT_PART_MAX_BYTES` per part (defaults
+   100k / 10 MB), whichever trips first; every part carries the header row so
+   it is a valid CSV alone, and header bytes count toward the part cap.  The
+   download route returns the ordered presigned-URL list
+   (`{part, url, rows, filename}` — `query-export-part-NNNN.csv`, signed with
+   `ResponseContentDisposition`, lifetime `EXPORT_PRESIGN_TTL_SECONDS` default
+   300s).  A single streamed row larger than the part cap is written anyway
+   (a row cannot be split); the total budget backstops that pathology.
+
+5. **Object layout + commit point.**  Under
+   `{EXPORT_S3_PREFIX}/{tenant_hash}/{snapshot_id}/`: `rows.jsonl.gz` (gzip
+   JSON Lines, one JSON array of REST-wire values per line — paging source of
+   truth), `data.part-NNNN.csv` part(s), then `meta.json` written LAST (the
+   commit point: a snapshot only "exists" once meta.json lands).  `meta.json`
+   carries `{snapshot_id, tenant_hash, columns: [{name, type}], row_count,
+   created_at, expires_at, parts, sql_sha256}` — the SQL text is NEVER stored,
+   only its sha256, and meta.json is never exposed raw to callers.  Staging is
+   local temp files (uploads start only after the stream completes), so an
+   aborted mint leaves no orphan objects.
+
+6. **Tenant binding = tenant-hash equality; expiry = app-side clock.**  The
+   `tenant_hash` is the full sha256 of canonical JSON (sorted keys) of the
+   per-request tenant settings dict (`clickhouse_client.tenant_settings()`).
+   Page/download recompute it from the CURRENT request and demand equality —
+   a foreign-tenant id returns 404 `SNAPSHOT_NOT_FOUND`, wire-indistinguishable
+   from an unknown id (existence is never probeable across tenants).  TTL
+   (`EXPORT_TTL_SECONDS`, default 4h) is enforced app-side on every read → 410
+   `SNAPSHOT_EXPIRED`; the bucket lifecycle rule (~1 day) is only the storage
+   backstop.  Snapshot ids are server-minted UUIDv4 (never derived from SQL —
+   refresh mints a fresh id) and are shape-checked before use in object keys.
+   RLS/row-policy freshness is evaluated once at MINT time
+   (`ensure_employee_access_fresh`); page/download verify only tenant hash +
+   expiry.  **Flagged for security review.**
+
+7. **Exact counts end the trailing empty fetch.**  Page responses are the
+   existing compact shape PLUS `row_count`, `snapshot_id`, `expires_at`;
+   `has_more` is EXACT (`offset + len(rows) < row_count`), and
+   `offset >= row_count` is a 200 with `rows: []`, not an error.
+
+8. **Value fidelity.**  Rows are NUL-padding-cleaned exactly like the MCP path
+   (`rstrip('\x00')` on strings incl. nested) and encoded with FastAPI
+   `jsonable_encoder` semantics (datetime/date → ISO strings, Decimal →
+   number, tuples → arrays) so snapshot rows are value-identical to what REST
+   `/query` returns today.  CSV carries canonical values ("" for null,
+   `true`/`false` for booleans, compact JSON for composites); display
+   formatting stays UI-side.
+
+9. **Storage + SDK.**  On-prem S3-compatible targets (Dell ECS / NetApp
+   StorageGRID): pinned `boto3==1.43.33`, path-style addressing (mandatory on
+   those appliances), SigV4 presigning (botocore presigns legacy SigV2 unless
+   pinned), credentials from env or Vault (`EXPORT_S3_CREDS_PATH`, keys
+   `EXPORT_S3_ACCESS_KEY` / `EXPORT_S3_SECRET_KEY` — the SCRATCH_CH vault
+   group pattern; least-privilege PUT/GET on the prefix).  boto3's default
+   managed checksum behavior is left untouched (vendor-supported default).  With
+   `EXPORT_ENABLED=true`, an incomplete S3 identity fails startup LOUD in the
+   lifespan (`export_s3_missing_fields()`), mirroring the auth fail-closed
+   pattern.
+
+10. **Streaming execution.**  `execute_query_rows_stream` keeps the
+    clickhouse-connect `QueryResult` (column names + types) while iterating
+    rows lazily — the public `query_rows_stream` discards it — and reuses
+    `_append_settings_clause` with the exact same tenant/CHProxy flag logic as
+    `execute_query` (RLS `rls_*` binds + caps merge into one SQL-body SETTINGS
+    clause; `readonly` never enters the SQL body; literal-`%` escaping
+    unchanged).
+
+---
+
+## Consequences
+
+**Positive:**
+- UI and export always agree (same frozen bytes); refresh is the only data
+  change vector; exact totals from first paint.
+- One expensive scan per answer instead of one per scroll page.
+- No silent truncation anywhere: over-budget mints fail with a caller-fixable
+  400; downloads always carry the full result set.
+- Zero regression risk to the interactive path: new routes, new caps, new
+  config — all off by default (`EXPORT_ENABLED=false`).
+
+**Negative / risks:**
+- Snapshots are server-side state with a TTL: storage cost lives on the
+  bucket until lifecycle reaps; an abandoned snapshot set is bounded by the
+  prefix + lifecycle rule, and app-side cleanup on abort is best-effort.
+- RLS freshness frozen at mint (decision 6) means a permissions change during
+  the TTL window does not affect an existing snapshot — flagged for security
+  review; TTL bounds the window.
+- Paging a row at offset N costs O(N) stream-decompressed reads of
+  `rows.jsonl.gz` (no index file); the 1M-row budget bounds this, and offsets
+  arrive in increasing scroll order in practice.
+- The export path's CH `max_result_rows = EXPORT_TOTAL_MAX_ROWS + 1` coupling
+  (decision 3) is load-bearing for the error contract: keep the one-row
+  headroom if the bundle is ever retuned.
+
+---
+
+## Alternatives Considered
+
+**Interactive-path reuse with bigger caps (rewrite `/query` limits):** Rejected.
+Weakens the interactive guardrails for a non-interactive use case and still
+re-executes the query per page (no UI/export consistency).
+
+**Server-side cursor / ClickHouse named sessions:** Rejected. Stateful
+ClickHouse sessions preclude horizontal scaling and leak between deploys;
+object storage is the cheap, HA home for frozen result sets.
+
+**TOO_LARGE detection via ClickHouse error text:** Rejected. Message strings
+are not a stable contract across CH versions; the budget must be enforced
+app-side with deterministic row/byte counters.
+
+---

+## Addendum (V4-0101010 follow-up): pluggable snapshot store — Couchbase default
+
+**Date:** 2026-10-23 · Amends decisions 5 (object layout) and 9 (storage + SDK)
+and §1's download transport.  S3 mode is RETAINED unchanged behind
+`EXPORT_STORE=s3`; everything else in this ADR still holds.
+
+1. **Pluggable store, Couchbase the default.**  `EXPORT_STORE=couchbase|s3`
+   (default `couchbase`) selects where snapshot artifacts live.  The layout,
+   split math, caps, budgets, tenant-hash binding, app-side TTL, and guardrail
+   chain are IDENTICAL in both modes; only the storage mechanics and the
+   download transport differ.  Both SDKs are LAZY-loaded — an s3-mode deployment
+   never imports the couchbase SDK and a couchbase-mode deployment never
+   imports boto3.
+
+2. **Couchbase doc layout** (dedicated bucket, doc ids shown for
+   `snap::{tenant_hash}::{snapshot_id}::`):
+   - `meta` — JSON object, same fields as `meta.json` (+ `chunk_rows`);
+     written LAST (commit point).
+   - `rows.{chunk:04d}` — JSON array of wire rows, 0-indexed,
+     `EXPORT_CB_CHUNK_ROWS` (default 10_000) rows per chunk; written DURING
+     streaming.  Paging fetches chunk doc(s) by `offset // chunk_rows`
+     ceil-math and slices in memory — O(1) doc reads, no stream-scan.
+   - `csv.{part:04d}` — binary CSV part bytes, 1-indexed, same part-split math
+     as S3 mode.
+   Every write carries a per-doc `expiry` of `EXPORT_TTL_SECONDS`; the app-side
+   TTL check stays authoritative.
+
+3. **Download transport in Couchbase mode.**  There is nothing to presign, so
+   `GET /query/snapshots/{id}/download-urls` returns RELATIVE app paths
+   (`query/snapshots/{id}/download?part=N`) with `expires_in: null`, and a new
+   route `GET /query/snapshots/{id}/download?part=N` streams the stored part
+   bytes as `text/csv` with a `Content-Disposition` attachment filename.
+   Per-request JWT authz (plus the usual tenant-hash/expiry checks) replaces
+   the presign window; an out-of-range part is 404 `SNAPSHOT_NOT_FOUND`.
+
+4. **Startup gate.**  Only the ACTIVE store's identity is gated:
+   `EXPORT_ENABLED=true` + `couchbase` requires conn string + bucket + username
+   + password; + `s3` keeps the existing four-field gate.  Credentials follow
+   the Vault group pattern (`EXPORT_CB_CREDS_PATH` → `EXPORT_CB_USERNAME` /
+   `EXPORT_CB_PASSWORD`), mirroring `EXPORT_S3_CREDS_PATH`.
+
+5. **Ops note (Couchbase provisioning).**  Provision a DEDICATED bucket for
+   snapshot docs (e.g. `agent_snapshots`): `replicas=0` (ephemeral cache-class
+   data), bucket **maxTTL = 14400s** (= `EXPORT_TTL_SECONDS`) as the
+   datastore-level TTL backstop under the per-doc expiry.  No query index is
+   required — all access is KV (get/upsert/remove + prefix scan for cleanup).
+
+**Why Couchbase by default:** the snapshot set is ephemeral KV-shaped state
+(id → bounded bytes) with a hard TTL; a KV bucket with per-doc expiry is
+operationally lighter than presigned object-store plumbing (no signature
+machinery on the hop chain) and keeps the download under per-request authz
+end-to-end.  S3 mode remains available where an object store is preferred.
+
+**Addendum (2026-10-23):** the populated Vault record at `EXPORT_CB_CREDS_PATH`
+carries the ops key names `COUCHBASE_USERNAME` / `COUCHBASE_PASSWORD` /
+`COUCHBASE_CONNECTION_STRING` / `COUCHBASE_BUCKET_EXPORT`, so the loader reads
+those (bucket always from `COUCHBASE_BUCKET_EXPORT`, never `COUCHBASE_BUCKET`);
+the `EXPORT_CB_*` names remain the env-var surface only.

+class FakeSnapshotStore:
+    """In-memory stand-in for S3SnapshotStore — records every interaction."""
+
+    def __init__(self) -> None:
+        self.objects: dict[str, bytes] = {}
+        self.uploads: list[str] = []          # upload order (commit-point check)
+        self.deleted_prefixes: list[str] = []
+        self.presign_calls: list[dict] = []
+
+    def put_file(self, local_path: str, key: str) -> None:
+        with open(local_path, "rb") as fh:
+            self.objects[key] = fh.read()
+        self.uploads.append(key)
+
+    def read_bytes(self, key: str) -> bytes:
+        if key not in self.objects:
+            raise SnapshotObjectMissingError(key)
+        return self.objects[key]
+
+    def open_reader(self, key: str):
+        if key not in self.objects:
+            raise SnapshotObjectMissingError(key)
+        return io.BytesIO(self.objects[key])
+
+    def delete_prefix(self, prefix: str) -> None:
+        self.deleted_prefixes.append(prefix)
+        for key in [k for k in self.objects if k.startswith(prefix)]:
+            del self.objects[key]
+
+    def presign_get(self, key: str, *, filename: str, expires_in: int) -> str:
+        self.presign_calls.append({"key": key, "filename": filename, "expires_in": expires_in})
+        return f"https://s3.test/b/{key}?sig=fake&expires={expires_in}"

+class FakeCouchbaseStore:
+    """Dict-backed stand-in for CouchbaseSnapshotStore (records write order)."""
+
+    def __init__(self) -> None:
+        self.docs: dict[str, bytes] = {}
+        self.writes: list[str] = []  # write order (commit-point check)
+        self.deleted_prefixes: list[str] = []
+
+    def put_json(self, key: str, value: Any) -> None:
+        self.docs[key] = json.dumps(value, separators=(",", ":")).encode("utf-8")
+        self.writes.append(key)
+
+    def put_file(self, local_path: str, key: str) -> None:
+        with open(local_path, "rb") as fh:
+            self.docs[key] = fh.read()
+        self.writes.append(key)
+
+    def read_bytes(self, key: str) -> bytes:
+        if key not in self.docs:
+            raise SnapshotObjectMissingError(key)
+        return self.docs[key]
+
+    def open_reader(self, key: str):
+        if key not in self.docs:
+            raise SnapshotObjectMissingError(key)
+        return io.BytesIO(self.docs[key])
+
+    def delete_prefix(self, prefix: str) -> None:
+        self.deleted_prefixes.append(prefix)
+        for key in [k for k in self.docs if k.startswith(prefix)]:
+            del self.docs[key]

+# ===========================================================================
+# 17. inject_limit=False — the answer-snapshot export path
+#
+# validate_and_sanitize(sql, default_limit, *, inject_limit=False) skips ONLY
+# the auto-LIMIT injection (snapshot exports materialize the FULL result set);
+# every other guardrail must stay byte-identical: comment stripping (and its
+# rejections), the multi-statement rejection, the allowlist, the denylist, and
+# all other output transformations.
+# ===========================================================================
+
+class TestInjectLimitFalse:
+    """Export-path validation: no LIMIT injected, every other guardrail intact."""
+
+    def test_no_limit_injected_when_disabled(self):
+        result = validate_and_sanitize("SELECT id FROM t", DEFAULT_LIMIT, inject_limit=False)
+        assert result == "SELECT id FROM t"
+
+    def test_with_query_no_limit_injected_when_disabled(self):
+        result = validate_and_sanitize(
+            "WITH x AS (SELECT 1) SELECT * FROM x", DEFAULT_LIMIT, inject_limit=False
+        )
+        assert "LIMIT" not in result
+
+    def test_existing_limit_preserved_when_disabled(self):
+        result = validate_and_sanitize(
+            "SELECT id FROM t LIMIT 42", DEFAULT_LIMIT, inject_limit=False
+        )
+        assert result == "SELECT id FROM t LIMIT 42"
+
+    def test_default_and_explicit_true_still_inject(self):
+        # Positional-call parity: the existing call sites are untouched.
+        assert validate_and_sanitize("SELECT 1", DEFAULT_LIMIT) == "SELECT 1 LIMIT 1000"
+        assert validate_and_sanitize("SELECT 1", DEFAULT_LIMIT, inject_limit=True) == (
+            "SELECT 1 LIMIT 1000"
+        )
+
+    def test_allowlist_still_enforced_when_disabled(self):
+        with pytest.raises(HTTPException) as exc_info:
+            validate_and_sanitize("INSERT INTO t VALUES (1)", DEFAULT_LIMIT, inject_limit=False)
+        assert exc_info.value.detail["code"] == "DISALLOWED_STATEMENT_TYPE"
+
+    def test_denylist_still_enforced_when_disabled(self):
+        with pytest.raises(HTTPException) as exc_info:
+            validate_and_sanitize(
+                "SELECT * FROM s3('https://example.invalid/f','CSV','x String')",
+                DEFAULT_LIMIT,
+                inject_limit=False,
+            )
+        assert exc_info.value.detail["code"] == "DISALLOWED_KEYWORD"
+
+    def test_user_settings_clause_still_enforced_when_disabled(self):
+        with pytest.raises(HTTPException) as exc_info:
+            validate_and_sanitize("SELECT 1 SETTINGS readonly=0", DEFAULT_LIMIT, inject_limit=False)
+        assert exc_info.value.detail["code"] == "DISALLOWED_KEYWORD"
+
+    def test_comment_handling_unchanged_when_disabled(self):
+        # Comments still stripped; unterminated comments still rejected.
+        result = validate_and_sanitize(

+        assert "LIMIT" not in result
+        with pytest.raises(HTTPException) as exc_info:

+        assert exc_info.value.detail["code"] == "INVALID_COMMENT"

+    def test_multi_statement_still_rejected_when_disabled(self):

+    def test_trailing_semicolon_still_stripped_when_disabled(self):
